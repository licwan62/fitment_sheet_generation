二、Power Query 主代码

这个查询的目标是输出一张 回填检查表，不会直接覆盖原表。

powerquery
let
    // 1. 读取当前工作簿里的表
    源全量表 = Excel.CurrentWorkbook(){[Name="全量表"]}[Content],
    源标准化表 = Excel.CurrentWorkbook(){[Name="车型标准化表"]}[Content],
    源代际映射表 = Excel.CurrentWorkbook(){[Name="代际映射表"]}[Content],

    // 2. 类型处理
    全量表_类型 = Table.TransformColumnTypes(
        源全量表,
        {
            {"主车型", type text},
            {"年份区间", type text},
            {"代际", type text},
            {"代际说明", type text}
        }
    ),

    标准化表_类型 = Table.TransformColumnTypes(
        源标准化表,
        {
            {"原主车型", type text},
            {"标准主车型", type text}
        }
    ),

    代际映射表_类型 = Table.TransformColumnTypes(
        源代际映射表,
        {
            {"标准主车型", type text},
            {"代际", type text},
            {"年份区间", type text},
            {"year_start", Int64.Type},
            {"year_end", Int64.Type},
            {"标准代际说明", type text},
            {"状态", type text},
            {"依据备注", type text}
        }
    ),

    // 3. 给全量表添加原始行号，方便后面回压
    添加原始行号 = Table.AddIndexColumn(
        全量表_类型,
        "原始行号",
        1,
        1,
        Int64.Type
    ),

    // 4. 合并车型标准化表
    合并标准化 = Table.NestedJoin(
        添加原始行号,
        {"主车型"},
        标准化表_类型,
        {"原主车型"},
        "标准化匹配",
        JoinKind.LeftOuter
    ),

    展开标准化 = Table.ExpandTableColumn(
        合并标准化,
        "标准化匹配",
        {"标准主车型"},
        {"标准主车型_匹配"}
    ),

    添加标准主车型 = Table.AddColumn(
        展开标准化,
        "标准主车型",
        each
            if [标准主车型_匹配] = null or Text.Trim([标准主车型_匹配]) = "" then
                [主车型]
            else
                [标准主车型_匹配],
        type text
    ),

    删除标准化临时列 = Table.RemoveColumns(
        添加标准主车型,
        {"标准主车型_匹配"}
    ),

    // 5. 解析年份区间为年份列表
    添加年份列表 = Table.AddColumn(
        删除标准化临时列,
        "年份列表",
        each
            let
                txt = Text.Trim(Text.From([年份区间])),
                parts = List.Select(
                    List.Transform(
                        Text.Split(txt, "/"),
                        each Text.Trim(_)
                    ),
                    each _ <> ""
                ),
                yearsNested =
                    List.Transform(
                        parts,
                        each
                            let
                                p = _,
                                hasRange = Text.Contains(p, "-"),
                                ys =
                                    if hasRange then
                                        Number.FromText(Text.BeforeDelimiter(p, "-"))
                                    else
                                        Number.FromText(p),
                                ye =
                                    if hasRange then
                                        Number.FromText(Text.AfterDelimiter(p, "-"))
                                    else
                                        ys,
                                yearList =
                                    if ys <= ye then
                                        {ys..ye}
                                    else
                                        {}
                            in
                                yearList
                    ),
                years = List.Sort(List.Distinct(List.Combine(yearsNested)))
            in
                years,
        type list
    ),

    // 6. 展开成逐年
    展开年份 = Table.ExpandListColumn(
        添加年份列表,
        "年份列表"
    ),

    重命名年份列 = Table.RenameColumns(
        展开年份,
        {{"年份列表", "年份"}}
    ),

    更改年份类型 = Table.TransformColumnTypes(
        重命名年份列,
        {{"年份", Int64.Type}}
    ),

    // 7. 按 标准主车型 + 年份 匹配代际映射表
    添加代际匹配表 = Table.AddColumn(
        更改年份类型,
        "代际匹配",
        each
            let
                car = [标准主车型],
                y = [年份],
                matched = Table.SelectRows(
                    代际映射表_类型,
                    each
                        [标准主车型] = car and
                        [year_start] <= y and
                        [year_end] >= y
                )
            in
                matched,
        type table
    ),

    // 8. 提取逐年匹配结果
    添加逐年代际 = Table.AddColumn(
        添加代际匹配表,
        "逐年代际",
        each
            let
                t = [代际匹配]
            in
                if Table.RowCount(t) = 1 then
                    t{0}[代际]
                else if Table.RowCount(t) = 0 then
                    "未匹配"
                else
                    "多重匹配",
        type text
    ),

    添加逐年代际说明 = Table.AddColumn(
        添加逐年代际,
        "逐年代际说明",
        each
            let
                t = [代际匹配]
            in
                if Table.RowCount(t) = 1 then
                    t{0}[标准代际说明]
                else
                    "",
        type text
    ),

    添加逐年状态 = Table.AddColumn(
        添加逐年代际说明,
        "逐年映射状态",
        each
            let
                t = [代际匹配]
            in
                if Table.RowCount(t) = 1 then
                    t{0}[状态]
                else if Table.RowCount(t) = 0 then
                    "未匹配"
                else
                    "多重匹配",
        type text
    ),

    // 9. 回压到原始行
    按原始行分组 = Table.Group(
        添加逐年状态,
        {"原始行号"},
        {
            {"原记录", each Table.FirstN(_, 1), type table},
            {"匹配年份", each Text.Combine(List.Transform(List.Sort(List.Distinct([年份])), Text.From), "/"), type text},
            {"匹配代际列表", each List.Sort(List.Distinct([逐年代际])), type list},
            {"匹配代际说明列表", each List.Sort(List.Distinct(List.Select([逐年代际说明], each _ <> ""))), type list},
            {"匹配状态列表", each List.Sort(List.Distinct([逐年映射状态])), type list}
        }
    ),

    展开原记录 = Table.ExpandTableColumn(
        按原始行分组,
        "原记录",
        Table.ColumnNames(删除标准化临时列),
        Table.ColumnNames(删除标准化临时列)
    ),

    // 10. 生成最终映射结果
    添加映射代际 = Table.AddColumn(
        展开原记录,
        "映射代际",
        each
            let
                xs = [匹配代际列表],
                clean = List.RemoveItems(xs, {"未匹配", "多重匹配"})
            in
                if List.Contains(xs, "多重匹配") then
                    "多重匹配"
                else if List.Contains(xs, "未匹配") and List.Count(clean) = 0 then
                    "未匹配"
                else if List.Count(clean) = 1 and not List.Contains(xs, "未匹配") then
                    clean{0}
                else if List.Count(clean) = 1 and List.Contains(xs, "未匹配") then
                    clean{0} & " / 部分年份未匹配"
                else
                    "跨代际，需要拆分",
        type text
    ),

五、跨代际自动拆分建议表

如果你想让 Power Query 自动生成“建议拆分后的年份区间”，可以新建一个查询，引用之前逐年展开后的中间结果。

如果你的主查询没有保留逐年表，建议单独做一个查询叫：

逐年代际匹配表

核心逻辑如下：

powerquery
let
    源 = 代际回填检查表,

    // 只取跨代际行
    筛选跨代际 = Table.SelectRows(
        源,
        each [代际映射状态] = "跨代际，需要拆分"
    ),

    // 重新展开匹配年份
    添加年份列表 = Table.AddColumn(
        筛选跨代际,
        "年份列表",
        each List.Transform(
            Text.Split([匹配年份], "/"),
            each Number.FromText(_)
        ),
        type list
    ),

    展开年份 = Table.ExpandListColumn(
        添加年份列表,
        "年份列表"
    ),

    重命名年份 = Table.RenameColumns(
        展开年份,
        {{"年份列表", "年份"}}
    ),

    更改类型 = Table.TransformColumnTypes(
        重命名年份,
        {{"年份", Int64.Type}}
    ),

    // 重新匹配代际映射表
    源代际映射表 = Excel.CurrentWorkbook(){[Name="代际映射表"]}[Content],

    代际映射表_类型 = Table.TransformColumnTypes(
        源代际映射表,
        {
            {"标准主车型", type text},
            {"代际", type text},
            {"year_start", Int64.Type},
            {"year_end", Int64.Type},
            {"标准代际说明", type text}
        }
    ),

    添加逐年代际 = Table.AddColumn(
        更改类型,
        "拆分后代际",
        each
            let
                car = [标准主车型],
                y = [年份],
                matched = Table.SelectRows(
                    代际映射表_类型,
                    each
                        [标准主车型] = car and
                        [year_start] <= y and
                        [year_end] >= y
                )
            in
                if Table.RowCount(matched) = 1 then
                    matched{0}[代际]
                else
                    "未匹配",
        type text
    ),

    添加逐年说明 = Table.AddColumn(
        添加逐年代际,
        "拆分后代际说明",
        each
            let
                car = [标准主车型],
                y = [年份],
                matched = Table.SelectRows(
                    代际映射表_类型,
                    each
                        [标准主车型] = car and
                        [year_start] <= y and
                        [year_end] >= y
                )
            in
                if Table.RowCount(matched) = 1 then
                    matched{0}[标准代际说明]
                else
                    "",
        type text
    ),

    // 按 原始行号 + 代际 + 代际说明 分组
    分组 = Table.Group(
        添加逐年说明,
        {
            "原始行号",
            "主车型",
            "标准主车型",
            "拆分后代际",
            "拆分后代际说明"
        },
        {
            {
                "年份列表",
                each List.Sort(List.Distinct([年份])),
                type list
            }
        }
    ),

    // 把连续年份压缩成年份区间
    添加建议年份区间 = Table.AddColumn(
        分组,
        "建议年份区间",
        each
            let
                years = [年份列表],
                grouped =
                    List.Accumulate(
                        years,
                        {},
                        (state, current) =>
                            if List.Count(state) = 0 then
                                {{current}}
                            else
                                let
                                    lastGroup = List.Last(state),
                                    lastYear = List.Last(lastGroup)
                                in
                                    if current = lastYear + 1 then
                                        List.RemoveLastN(state, 1) & {lastGroup & {current}}
                                    else
                                        state & {{current}}
                    ),
                textList =
                    List.Transform(
                        grouped,
                        each
                            if List.Count(_) = 1 then
                                Text.From(_{0})
                            else
                                Text.From(List.First(_)) & "-" & Text.From(List.Last(_))
                    )
            in
                Text.Combine(textList, "/"),
        type text
    ),

    删除年份列表 = Table.RemoveColumns(
        添加建议年份区间,
        {"年份列表"}
    ),

    调整列顺序 = Table.ReorderColumns(
        删除年份列表,
        {
            "原始行号",
            "主车型",
            "标准主车型",
            "建议年份区间",
            "拆分后代际",
            "拆分后代际说明"
        }
    )
in
    调整列顺序

七、最终建议执行顺序
tsv
顺序	动作
1	跑 代际回填检查表
2	导出 可入库回填表
3	先处理 多重匹配表
4	再处理 未匹配表
5	最后处理 跨代际拆分表
6	确认异常清零后，再覆盖原代际/代际说明

实际优先级建议：

tsv
优先级	异常类型	原因
1	多重匹配	这是映射表自身错误，必须先修
2	未匹配	通常是主车型标准化或映射缺失
3	跨代际，需要拆分	这是原表记录需要拆行
4	待参考/待终核	人工确认后再入库

下一步可以做 “代际映射表自身质量检查”，专门检查映射表里有没有同车型年份重叠、年份断层、重复代际、状态异常。
