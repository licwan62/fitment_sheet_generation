三、Power Query 合并逻辑

在组合表 Power Query 里，用 品牌 + 主车型 去匹配规则表。

powerquery
let
    源 = 你的组合表上一步,

    规则表 = Excel.CurrentWorkbook(){[Name="特殊版本保留规则表"]}[Content],

    规则表类型 = Table.TransformColumnTypes(
        规则表,
        {
            {"品牌", type text},
            {"主车型", type text},
            {"强保留提示词", type text},
            {"弱保留提示词", type text},
            {"不保留提示词", type text},
            {"组合处理建议", type text}
        }
    ),

    合并规则 = Table.NestedJoin(
        源,
        {"品牌", "主车型"},
        规则表类型,
        {"品牌", "主车型"},
        "特殊版本规则",
        JoinKind.LeftOuter
    ),

    展开规则 = Table.ExpandTableColumn(
        合并规则,
        "特殊版本规则",
        {"强保留提示词", "弱保留提示词", "不保留提示词", "组合处理建议"},
        {"强保留提示词", "弱保留提示词", "不保留提示词", "组合处理建议"}
    ),

    添加组合版本提示 = Table.AddColumn(
        展开规则,
        "组合版本提示",
        each
            if [强保留提示词] <> null and Text.Trim([强保留提示词]) <> "" then
                "incl: " & [强保留提示词]
            else if [弱保留提示词] <> null and Text.Trim([弱保留提示词]) <> "" then
                "incl: " & [弱保留提示词]
            else
                "",
        type text
    )
in
    添加组合版本提示

            try Text.From([货斗长度_ft]) otherwise null
        }),
        " "
    ),
    type text
)

然后加“是否允许提示”：

powerquery
添加是否允许提示 = Table.AddColumn(
    添加匹配文本,
    "是否允许特殊版本提示",
    each
        let
            structRule = if [适用结构] = null then "" else Text.Trim([适用结构]),
            keywordRule = if [适用版本关键词] = null then "" else Text.Trim([适用版本关键词]),
            rowStruct = if [组合结构] = null then "" else Text.Trim([组合结构]),
            rowText = if [组合匹配文本] = null then "" else Text.Lower([组合匹配文本]),

            structOK =
                structRule = ""
                or Text.Contains(Text.Lower(structRule), Text.Lower(rowStruct))
                or Text.Contains(Text.Lower(rowStruct), Text.Lower(structRule)),

            keywordList =
                if keywordRule = "" then
                    {}
                else
                    List.Transform(Text.Split(keywordRule, "|"), each Text.Lower(Text.Trim(_))),

            keywordOK =
                List.Count(keywordList) = 0
                or List.AnyTrue(
                    List.Transform(
                        keywordList,
                        each Text.Contains(rowText, _)
                    )
                )
        in
            structOK and keywordOK,
    type logical
)

最后生成提示：

powerquery
添加组合版本提示 = Table.AddColumn(
    添加是否允许提示,
    "组合版本提示",
    each
        if [是否允许特殊版本提示] <> true then
            ""
        else if [强保留提示词] <> null and Text.Trim([强保留提示词]) <> "" then
            "incl: " & [强保留提示词]
        else if [弱保留提示词] <> null and Text.Trim([弱保留提示词]) <> "" then
            "incl: " & [弱保留提示词]
        else
            "",
    type text
)
更稳的落地建议

如果你现在的组合表还没有很细的 组合版本，那就先别加关键词限制，只做第一版：

tsv
阶段	做法
第一版	按 品牌+主车型 匹配规则表，生成粗提示
第二版	对 Pickup、911、Wrangler、Defender、长轴 SUV 这些高风险车型加适用结构/关键词限制
第三版	把提示词拆成多行规则，一条提示词一行，做到更精准匹配

我建议下一步把规则表从“每车型一行”改成一条提示词一行，这样 Power Query 会更好控制，比如：
